{{- else if contains "NodePort" .Values.service.type }}

Access LibreDB Studio via NodePort:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "libredb-studio.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "http://${NODE_IP}:${NODE_PORT}"

{{- else if contains "LoadBalancer" .Values.service.type }}

Access LibreDB Studio via LoadBalancer:
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  Watch status with: kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "libredb-studio.fullname" . }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "libredb-studio.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "http://${SERVICE_IP}:{{ .Values.service.port }}"

{{- if and (not (include "libredb-studio.authStrict" .)) (eq .Values.authProvider "local") (not .Values.secrets.adminPassword) (not .Values.secrets.existingSecret) }}

First-run credentials (zero-config bootstrap):
  secrets.adminPassword was not set, so LibreDB Studio generates admin
  credentials on first start and prints them once to the pod log. Retrieve them:

    kubectl --namespace {{ .Release.Namespace }} logs deployment/{{ include "libredb-studio.fullname" . }} | grep -A 4 "generated admin credentials"

  Nothing printed? The banner appears once, in the log of the container that ran
  the first start:
  - after a container restart, add --previous to the command above
  - or read the file the app stored them in:
      kubectl --namespace {{ .Release.Namespace }} exec deploy/{{ include "libredb-studio.fullname" . }} -- cat /app/data/auth-bootstrap.json
{{- if gt (int .Values.replicaCount) 1 }}
  - with {{ .Values.replicaCount }} replicas, `logs deployment/...` reads one pod
    chosen arbitrarily; list pods by start time and read the oldest one instead
{{- end }}

  - The default install is admin-only. Set secrets.userPassword to enable the
    second, non-admin user account.
{{- if include "libredb-studio.persistenceEnabled" . }}
  - Generated credentials are stored in the persistent data volume
    (/app/data/auth-bootstrap.json) and survive pod restarts and recreation.
{{- else }}
  - Generated credentials are stored in /app/data/auth-bootstrap.json on an
    emptyDir volume: they survive container restarts but are regenerated when
    the pod is recreated. For stable credentials set persistence.enabled=true
    or provide secrets.jwtSecret / secrets.adminPassword yourself.
{{- end }}
{{- end }}
